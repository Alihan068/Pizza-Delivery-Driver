using System.Collections.Generic;
using UnityEngine;

/// <summary>Optional bounded one-shot presentation pool. Missing effects never gate gameplay explosions.</summary>
public sealed class TrafficExplosionVisualPool : MonoBehaviour {
    [SerializeField] TrafficDamageWorld world;
    [SerializeField] ParticleSystem effectPrefab;
    [SerializeField, Min(0)] int capacity = 16;
    readonly List<ParticleSystem> effects = new List<ParticleSystem>();
    TrafficSessionCoordinator subscribedCoordinator;
    bool subscribed;

    void Start() { Initialize(world, effectPrefab, capacity); }

    /// <summary>
    /// Prewarms owned effects once and subscribes to visual requests. Safe before the damage world
    /// is configured: the first actual request binds the session-end callback before playing.
    /// </summary>
    public void Initialize(TrafficDamageWorld owner, ParticleSystem prefab, int limit) {
        if (subscribed || owner == null) return;
        world = owner;
        if (prefab != null) {
            for (int i = 0; i < Mathf.Max(0, limit); i++) {
                var effect = Instantiate(prefab, transform);
                effect.Stop(true, ParticleSystemStopBehavior.StopEmittingAndClear);
                effects.Add(effect);
            }
        }
        world.ExplosionVisualRequested += Show;
        BindCoordinator();
        subscribed = true;
    }

    void BindCoordinator() {
        if (subscribedCoordinator != null || world == null || world.Coordinator == null) return;
        subscribedCoordinator = world.Coordinator;
        subscribedCoordinator.Ended += StopEffects;
    }

    void Show(Vector2 position) {
        if (world == null || !world.IsActive) return;
        BindCoordinator();
        for (int i = 0; i < effects.Count; i++) {
            var effect = effects[i];
            if (effect == null || effect.IsAlive(true)) continue;
            effect.transform.position = position;
            effect.Play(true);
            return;
        }
    }

    void StopEffects() {
        foreach (var effect in effects)
            if (effect != null) effect.Stop(true, ParticleSystemStopBehavior.StopEmittingAndClear);
    }

    void OnDestroy() {
        if (subscribed && world != null) world.ExplosionVisualRequested -= Show;
        if (subscribedCoordinator != null) subscribedCoordinator.Ended -= StopEffects;
    }
}
