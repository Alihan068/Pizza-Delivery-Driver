using UnityEngine;

/// <summary>Optional pooled feedback; visual availability never controls gameplay damage.</summary>
public sealed class NpcDamagePresentation : MonoBehaviour {
    [SerializeField] SpriteRenderer body;
    [SerializeField] GameObject smoke;
    [SerializeField] GameObject critical;
    [SerializeField] Color wreckColor = Color.gray;
    Color originalColor;
    bool captured;
    ParticleSystem[] smokeParticles;
    ParticleSystem[] criticalParticles;

    /// <summary>Restores authored color and clears all particle state before a pooled life begins.</summary>
    public void ResetForLife() {
        CacheEffects();
        if (!captured && body != null) { originalColor = body.color; captured = true; }
        if (body != null && captured) body.color = originalColor;
        SetEffect(smoke, smokeParticles, false);
        SetEffect(critical, criticalParticles, false);
    }

    /// <summary>Applies resolved feedback without changing health; wrecks retain only their body sprite.</summary>
    public void Show(NpcDamageFeedbackLevel level, bool wreck) {
        CacheEffects();
        SetEffect(smoke, smokeParticles, !wreck && level != NpcDamageFeedbackLevel.None);
        SetEffect(critical, criticalParticles, !wreck && level == NpcDamageFeedbackLevel.Critical);
        if (wreck && body != null) body.color = wreckColor;
    }

    void CacheEffects() {
        if (smokeParticles == null) smokeParticles = smoke != null ? smoke.GetComponentsInChildren<ParticleSystem>(true) : System.Array.Empty<ParticleSystem>();
        if (criticalParticles == null) criticalParticles = critical != null ? critical.GetComponentsInChildren<ParticleSystem>(true) : System.Array.Empty<ParticleSystem>();
    }

    static void SetEffect(GameObject effect, ParticleSystem[] particles, bool enabled) {
        if (effect == null) return;
        bool wasActive = effect.activeSelf;
        if (enabled) {
            effect.SetActive(true);
            if (!wasActive) foreach (var particle in particles) if (particle != null) particle.Play(false);
        } else {
            foreach (var particle in particles) if (particle != null) particle.Stop(false, ParticleSystemStopBehavior.StopEmittingAndClear);
            effect.SetActive(false);
        }
    }
}
